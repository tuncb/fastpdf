cmake_minimum_required(VERSION 3.24)
project(FastPDF VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "FastPDF uses the Windows SDK. Build on Windows 10 or newer.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(FASTPDF_SANITIZE "Enable MSVC AddressSanitizer" OFF)
add_library(pdf_engine STATIC src/pdf.cpp src/filters.cpp src/security.cpp src/functions.cpp src/fonts.cpp src/font_data.cpp src/arithmetic.cpp src/jpeg2000.cpp src/jbig2.cpp src/fax.cpp src/images.cpp src/render.cpp)
target_include_directories(pdf_engine PUBLIC src)
target_compile_definitions(pdf_engine PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(pdf_engine PUBLIC /W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus)
target_link_libraries(pdf_engine PUBLIC d2d1 dwrite windowscodecs bcrypt ole32 gdi32 normaliz mscms)
if(FASTPDF_SANITIZE)
  target_compile_options(pdf_engine PUBLIC /fsanitize=address /Zi)
  target_link_options(pdf_engine PUBLIC /INCREMENTAL:NO)
else()
  target_compile_options(pdf_engine PUBLIC $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
  target_link_options(pdf_engine PUBLIC $<$<CONFIG:Release>:/LTCG>)
endif()
add_library(deployment STATIC src/deployment.cpp src/deployment_protocol.cpp)
target_link_libraries(deployment PUBLIC pdf_engine winhttp version shell32 shlwapi advapi32 uuid)
target_compile_definitions(deployment PUBLIC FASTPDF_VERSION="${PROJECT_VERSION}" FASTPDF_VERSION_W=L"${PROJECT_VERSION}")
add_executable(fastpdf WIN32 src/app.cpp src/installation_dialog.cpp src/app.rc src/app.manifest)
add_custom_target(build_metadata
  COMMAND ${CMAKE_COMMAND} "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
    "-DOUTPUT_DIR=${CMAKE_CURRENT_BINARY_DIR}" "-DVERSION=${PROJECT_VERSION}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-metadata.cmake"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/release-manifest.json"
  VERBATIM)
add_dependencies(fastpdf build_metadata)
set_source_files_properties(src/app.rc PROPERTIES
  OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/release-manifest.json;${CMAKE_CURRENT_SOURCE_DIR}/src/assets/fastpdf.ico")
target_include_directories(fastpdf PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/docs")
target_link_libraries(fastpdf PRIVATE deployment comctl32 comdlg32 shell32 dwmapi msimg32 uxtheme)
add_executable(pdfprobe src/cli.cpp src/console.manifest)
target_link_libraries(pdfprobe PRIVATE pdf_engine)
enable_testing()
add_executable(deployment_tests tests/deployment.cpp src/deployment.cpp src/deployment_protocol.cpp src/installation_dialog.cpp src/app.rc src/app.manifest)
target_include_directories(deployment_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/docs")
target_link_libraries(deployment_tests PRIVATE pdf_engine winhttp version shell32 shlwapi advapi32 uuid comctl32 dwmapi)
target_compile_definitions(deployment_tests PRIVATE FASTPDF_DEPLOYMENT_TEST
  FASTPDF_VERSION="${PROJECT_VERSION}" FASTPDF_VERSION_W=L"${PROJECT_VERSION}")
add_dependencies(deployment_tests fastpdf)
add_test(NAME deployment COMMAND deployment_tests $<TARGET_FILE:fastpdf> ${CMAKE_CURRENT_BINARY_DIR}/deployment-tests)
add_executable(pdf_tests tests/tests.cpp)
target_link_libraries(pdf_tests PRIVATE pdf_engine)
add_executable(codec_probe tests/codec_probe.cpp src/console.manifest)
target_link_libraries(codec_probe PRIVATE pdf_engine)
add_test(NAME engine COMMAND pdf_tests)
add_executable(render_resources_tests tests/render_resources.cpp)
target_link_libraries(render_resources_tests PRIVATE pdf_engine)
add_test(NAME render_resources COMMAND render_resources_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/data)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME probe_batch COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/probe_batch.py $<TARGET_FILE:pdfprobe> ${CMAKE_CURRENT_BINARY_DIR}/probe-batch)
  set_tests_properties(probe_batch PROPERTIES TIMEOUT 60)
  add_test(NAME reader_restore COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reader_restore.py $<TARGET_FILE:fastpdf> ${CMAKE_CURRENT_BINARY_DIR}/reader-restore)
  set_tests_properties(reader_restore PROPERTIES TIMEOUT 60)
  add_test(NAME fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:pdfprobe> ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
  set_tests_properties(fixtures PROPERTIES TIMEOUT 180)
  add_test(NAME codecs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/codec_regression.py $<TARGET_FILE:codec_probe> ${CMAKE_CURRENT_BINARY_DIR}/codec-results)
  set_tests_properties(codecs PROPERTIES TIMEOUT 180)
  add_test(NAME ui_toolbar COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_toolbar.py $<TARGET_FILE:fastpdf> ${CMAKE_CURRENT_BINARY_DIR}/ui-toolbar)
  set_tests_properties(ui_toolbar PROPERTIES TIMEOUT 60)
  add_test(NAME print_preview COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/print_preview.py $<TARGET_FILE:fastpdf> ${CMAKE_CURRENT_BINARY_DIR}/print-preview)
  set_tests_properties(print_preview PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
  add_test(NAME file_paths COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/file_paths.py $<TARGET_FILE:fastpdf> $<TARGET_FILE:pdfprobe> ${CMAKE_CURRENT_BINARY_DIR}/file-paths)
  set_tests_properties(file_paths PROPERTIES TIMEOUT 120)
endif()
install(TARGETS fastpdf RUNTIME DESTINATION .)
